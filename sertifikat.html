async function uploadPDFKeDriveDanWA() { const btn = document.querySelector('.btn-wa-kirim'); // Ganti dengan Web App URL (Deployment Baru) & Nomor WA Admin const urlWebAppGAS = "https://script.google.com/macros/s/AKfycbwG3rJUL5ILV62pJ7GczsoZ1alPkCbNqh_708JBtL4cQh--XMz44Og9vj3axqkZNpJl/exec"; const noAdminWA = "6285766594397"; const namaUser = document.getElementById("certNama").innerText || "Kontributor"; btn.disabled = true; btn.innerHTML = '⏳ Memproses Verifikasi...'; const elemen = document.getElementById("sertifikatBox"); // Render PDF ke Base64 const opsiPDF = { margin: 0, filename: `Sertifikat-CFP-${namaUser.replace(/\s+/g, '_')}.pdf`, image: { type: 'jpeg', quality: 0.95 }, html2canvas: { scale: 2, useCORS: true }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'landscape' } }; try { const pdfWorker = html2pdf().set(opsiPDF).from(elemen).outputPdf('datauristring'); const pdfDataUri = await pdfWorker; const pdfBase64Data = pdfDataUri.split(',')[1]; // Kirim ke Google Apps Script tanpa bentrok CORS const response = await fetch(urlWebAppGAS, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8', // Menghindari preflight CORS }, body: JSON.stringify({ filename: `Sertifikat_Agroforestry_${namaUser.replace(/\s+/g, '_')}.pdf`, pdfBase64: pdfBase64Data }) }); const resData = await response.json(); if (resData.result === "success") { const fileDriveUrl = resData.fileUrl; // Format Pesan ke Admin WA const pesanWA = `Halo Ethnicitizen,%0A%0ASaya telah mengajukan Sertifikat Carbon Footprint atas nama:*${encodeURIComponent(namaUser)}*.%0A%0A📁 *Link Dokumen PDF (Google Drive):*%0A${fileDriveUrl}%0A%0AMohon verifikasi pembayaran saya agar sertifikat tersebut terverifikasi sah.%0A%0ATerima kasih! 🌿`; window.open(`https://wa.me/${noAdminWA}?text=${pesanWA}`, '_blank'); btn.innerHTML = ' Berhasil Terkirim!'; } else { throw new Error(resData.error); } } catch (err) { console.error("Error Detail:", err); alert("Terjadi kendala saat memproses dan hubungkan langsung ke WhatsApp Admin..."); // Fallback jika terjadi error const pesanFallback = `Halo Ethnicitizen, saya ingin konfirmasi sertifikat atas nama *${encodeURIComponent(namaUser)}*.`; window.open(`https://wa.me/${noAdminWA}?text=${pesanFallback}`, '_blank'); } finally { btn.disabled = false; } }
