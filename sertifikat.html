async function simpanDanKirimSertifikat() { const element = document.getElementById('sertifikat-area'); // ID kontainer sertifikat Anda const btnSimpan = document.getElementById('btn-simpan'); // Tombol yang diklik // Ganti dengan Web App URL dari Google Apps Script Anda (Langkah 1) const URL_WEB_APP_GAS = "https://script.google.com/macros/s/AKfycbwNcBZIgrpVXT2xRshrZY8vukwIh2DyGJIW9uuugXGbRXWMTR7fkqMS4o6Zg0gn3B3S/exec"; try { // 1. Tampilkan indikator loading pada tombol btnSimpan.disabled = true; btnSimpan.innerText = "⏳ Membuat & Mengunggah Sertifikat..."; // 2. Konfigurasi html2pdf agar ukuran file kecil & tajam (di bawah 2MB) const opsiPDF = { margin: 0, filename: `Sertifikat_Karbon.pdf`, image: { type: 'jpeg', quality: 0.85 }, html2canvas: { scale: 1.5, useCORS: true }, // scale 1.5 aman & cepat jsPDF: { unit: 'mm', format: 'a4', orientation: 'landscape' } }; // 3. Render HTML ke bentuk Base64 Data String const pdfBase64 = await html2pdf().set(opsiPDF).from(element).outputPdf('datauristring'); // 4. Kirim ke Google Apps Script via POST (Content-Type: text/plain agar tidak terkena CORS block) const response = await fetch(URL_WEB_APP_GAS, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify({ filename: `Sertifikat_${Date.now()}.pdf`, pdfBase64: pdfBase64 }) }); const hasil = await response.json(); if (hasil.status === 'success') { alert("✅ Sertifikat berhasil disimpan ke Google Drive!"); console.log("Link PDF:", hasil.url); // Opsional: Buka WhatsApp Admin melampirkan Link Drive const pesanWA = `Halo Admin Ethnicitizen, saya telah menyelesaikan kompensasi karbon. Berikut sertifikat saya: ${hasil.url}`; window.open(`https://wa.me/628XXXXXXXXXX?text=${encodeURIComponent(pesanWA)}`, '_blank'); } else { throw new Error(hasil.message); } } catch (err) { console.error("Gagal mengunggah:", err); alert("❌ Terjadi kesalahan saat mengunggah sertifikat: " + err.message); } finally { btnSimpan.disabled = false; btnSimpan.innerText = "📜 Unduh / Kirim Sertifikat"; } }
